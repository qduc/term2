import net from 'node:net';import tls from 'node:tls';import http from 'node:http';import https from 'node:https';let network=0,ipc=0;const deny=()=>{network++;throw new Error('external network denied')};net.Socket.prototype.connect=function(...args){const o=Array.isArray(args[0])?args[0][0]:args[0];const p=typeof o==='string'?o:o?.path;if(typeof p==='string'&&p.startsWith('/')){ipc++;throw new Error('local IPC denied separately')}return deny()};tls.connect=deny;http.request=deny;http.get=deny;https.request=deny;https.get=deny;globalThis.fetch=deny;process.on('exit',()=>{console.error('EXTERNAL_NETWORK_ATTEMPTS='+network+' BLOCKED_LOCAL_IPC='+ipc);if(network)process.exitCode=1;});
