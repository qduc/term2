// Spawns one process per mutant; reports expected vs observed verdict.
import { spawnSync } from 'node:child_process';
import { readFileSync } from 'node:fs';
const here = new URL('.', import.meta.url).pathname;
const [runs = '300', seed = '20261024', list = 'mutants.json'] = process.argv.slice(2);
const mutants = JSON.parse(readFileSync(here + list, 'utf8'));
for (const m of mutants) {
  const r = spawnSync(process.execPath, ['--import', 'tsx', '--import', here + 'mutant-hook.mjs', here + 'probe.mjs',
    ...m.prop.split(' '), runs, seed], { env: { ...process.env, MUTANT_FILE: m.file ?? '', MUTANT_EDITS: JSON.stringify(m.edits) }, encoding: 'utf8' });
  let out; try { out = JSON.parse(r.stdout.trim().split('\n').at(-1)); } catch { out = { crash: (r.stderr.match(/Error.*$/m) ?? [r.stderr.slice(-200)])[0] }; }
  const observed = out.crash ? 'crash' : out.failed ? 'fail' : 'pass';
  console.log(JSON.stringify({ id: m.id, expect: m.expect, observed, ok: observed === m.expect, numRuns: out.numRuns, error: out.error ?? out.crash ?? null }));
}
