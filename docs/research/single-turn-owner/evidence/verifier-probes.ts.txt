it.each(['streaming', 'approval'])('closes a public turn observer during %s and permits its successor', async (mode) => {
  const root = await mkdtemp(join(tmpdir(), 'term2-owner-abandon-'));
  const providerId = 'single-owner-abandon-verifier';
  let requests = 0;
  let aborted = false;
  let dispatches = 0;
  registerProvider({
    id: providerId,
    label: 'Verifier',
    fetchModels: async () => [{ id: 'verifier-model' }],
    createStreamedModel: () => ({
      async *stream(request) {
        requests++;
        request.signal?.addEventListener('abort', () => { aborted = true; }, { once: true });
        yield { type: 'text_delta', text: requests === 1 ? 'BEFORE_ABANDON' : 'SUCCESSOR' };
        if (requests === 1) {
          if (mode === 'approval') {
            yield { type: 'tool_call', id: 'pending-abandon-call', name: 'verifier_pending', arguments: '{}' };
            return;
          }
          await new Promise<void>((_resolve, reject) => {
            const abort = () => {
              aborted = true;
              reject(Object.assign(new Error('Operation aborted'), { name: 'AbortError' }));
            };
            request.signal?.addEventListener('abort', abort, { once: true });
            if (request.signal?.aborted) abort();
          });
        }
        yield { type: 'completion', responseId: `verifier-${requests}`, output: [] };
      },
    }),
  });
  const settings = new SettingsService({ settingsDir: root, disableFilePersistence: true, disableLogging: true, env: {}, cli: {} });
  settings.set('agent.modelSelection', { provider: providerId, model: 'verifier-model' }, { persist: false });
  const context = new SessionContextService();
  const logger = new LoggingService({ disableLogging: true, logDir: root });
  const ownership = new ToolOwnershipRegistry();
  const client = new AgentClient({
    selection: { provider: providerId, model: 'verifier-model' },
    agentOverride: { name: 'Verifier', model: 'verifier-model', instructions: '', tools: [{ name: 'verifier_pending', description: 'Verifier pending', parameters: z.object({}), needsApproval: () => true, execute: async () => { dispatches++; return 'unexpected execution'; } }] },
    toolOwnership: ownership,
    deps: { settings, logger, sessionContextService: context },
  });
  const runtime = createSessionRuntime({ sessionId: 'abandon-verifier', agentClient: client, toolOwnership: ownership, deps: { logger, settingsService: settings, sessionContextService: context } });
  try {
    const observer = runtime.turns.start('first')[Symbol.asyncIterator]();
    const event = await observer.next();
    expect(event.done).toBe(false);
    if (mode === 'approval') await vi.waitFor(() => expect(runtime.approval.getPending()).toBeTruthy());
    const returned = await Promise.race([
      observer.return?.(),
      new Promise<never>((_resolve, reject) => setTimeout(() => reject(new Error('observer return timed out')), 2000)),
    ]);
    expect(returned?.done).toBe(true);
    if (mode === 'streaming') expect(aborted).toBe(true);
    const successor = [];
    for await (const next of runtime.turns.start('second')) successor.push(next);
    expect(JSON.stringify(successor)).toContain('SUCCESSOR');
    expect(requests).toBe(2);
    expect(dispatches).toBe(0);
  } finally {
    client.abort();
    runtime.dispose();
    unregisterProvider(providerId);
    await rm(root, { recursive: true, force: true });
  }
});

it('keeps the same public turn alive through manual approval then a real post-execute gate', async () => {
  const root = await mkdtemp(join(tmpdir(), 'term2-owner-postexecute-'));
  const providerId = 'single-owner-postexecute-verifier';
  const requests: unknown[] = [];
  registerProvider({
    id: providerId,
    label: 'Verifier',
    fetchModels: async () => [{ id: 'verifier-model' }],
    createStreamedModel: () => ({
      async *stream(request) {
        requests.push(request.input);
        if (requests.length === 1) {
          yield { type: 'tool_call', id: 'postexecute-call', name: 'verifier_postexecute', arguments: '{}' };
          return;
        }
        yield { type: 'text_delta', text: 'POSTEXECUTE_SUCCESS' };
        yield { type: 'completion', responseId: 'postexecute-response', output: [] };
      },
    }),
  });
  const settings = new SettingsService({ settingsDir: root, disableFilePersistence: true, disableLogging: true, env: {}, cli: {} });
  settings.set('agent.modelSelection', { provider: providerId, model: 'verifier-model' }, { persist: false });
  settings.set('shell.autoApproveMode', 'off', { persist: false });
  const context = new SessionContextService();
  const logger = new LoggingService({ disableLogging: true, logDir: root });
  const ownership = new ToolOwnershipRegistry();
  const policies = new ToolApprovalPolicyRegistry();
  const pending = new PostExecutePendingRegistry({ sessionId: 'postexecute-verifier', epoch: 1 });
  const capability = new PostExecutePauseCapability(pending);
  let executions = 0;
  const tools = buildAgentTools({
    toolDefinitions: [{
      name: 'verifier_postexecute', description: 'Verifier', parameters: z.object({}), needsApproval: () => true,
      execute: async () => { executions++; await writeFile(join(root, 'effects.txt'), String(executions)); return executions === 1 ? 'UNAPPROVED_RESULT' : 'APPROVED_RESULT'; },
      postExecutePause: { describe: () => ({ toolName: 'verifier_postexecute', argumentsText: '{}' }) },
    }],
    resolvedModel: 'verifier-model', shouldUseNativePatchTool: false,
    deps: { settings, logger, providerId, checkToolInterceptors: async () => null, approvalPolicyRegistry: policies, postExecutePauseCapability: capability } as AgentFactoryDeps,
  });
  const client = new AgentClient({
    selection: { provider: providerId, model: 'verifier-model' },
    agentOverride: { name: 'Verifier', model: 'verifier-model', instructions: '', tools },
    approvalPolicyRegistry: policies, toolOwnership: ownership,
    deps: { settings, logger, sessionContextService: context },
  });
  const runtime = createSessionRuntime({ sessionId: 'postexecute-verifier', agentClient: client, toolOwnership: ownership, approvalPolicyRegistry: policies, postExecutePending: pending, postExecutePauseCapability: capability, deps: { logger, settingsService: settings, sessionContextService: context } });
  try {
    const initial = [];
    for await (const event of runtime.turns.start('exercise gate')) initial.push(event);
    expect(JSON.stringify(initial)).toContain('approval_required');
    expect(executions).toBe(0);
    const afterApproval = [];
    for await (const event of runtime.turns.continueAfterApproval({ answer: 'y' })) afterApproval.push(event);
    expect(JSON.stringify(afterApproval)).toContain('approval_required');
    expect(executions).toBe(1);
    expect(requests).toHaveLength(1);
    const snapshot = pending.snapshot();
    expect(snapshot.entries).toHaveLength(1);
    expect(runtime.approval.decidePostExecutePending({ revision: snapshot.revision, ids: snapshot.entries.map((entry) => entry.id), decision: 'approve' })).toMatchObject({ kind: 'settled' });
    const settled = [];
    for await (const event of runtime.turns.continueAfterPostExecuteApproval()) settled.push(event);
    expect(JSON.stringify(settled)).toContain('POSTEXECUTE_SUCCESS');
    expect(executions).toBe(2);
    expect(await readFile(join(root, 'effects.txt'), 'utf8')).toBe('2');
    expect(requests).toHaveLength(2);
    expect(JSON.stringify(requests[1])).toContain('APPROVED_RESULT');
    expect(pending.snapshot().entries).toHaveLength(0);
  } finally {
    client.abort(); runtime.dispose(); unregisterProvider(providerId); await rm(root, { recursive: true, force: true });
  }
});

it('cancels a public turn during retry backoff without dispatch and permits a successor', async () => {
  const root = await mkdtemp(join(tmpdir(), 'term2-owner-retry-'));
  const providerId = 'single-owner-retry-verifier';
  let requests = 0;
  registerProvider({
    id: providerId, label: 'Verifier', fetchModels: async () => [{ id: 'verifier-model' }],
    createStreamedModel: () => ({
      async *stream() {
        requests++;
        if (requests === 1) throw Object.assign(new Error('ECONNRESET'), { code: 'ECONNRESET' });
        yield { type: 'text_delta', text: 'RETRY_SUCCESSOR' };
        yield { type: 'completion', responseId: 'retry-successor', output: [] };
      },
    }),
  });
  const settings = new SettingsService({ settingsDir: root, disableFilePersistence: true, disableLogging: true, env: {}, cli: {} });
  settings.set('agent.modelSelection', { provider: providerId, model: 'verifier-model' }, { persist: false });
  const context = new SessionContextService();
  const logger = new LoggingService({ disableLogging: true, logDir: root });
  const ownership = new ToolOwnershipRegistry();
  const client = new AgentClient({
    selection: { provider: providerId, model: 'verifier-model' },
    agentOverride: { name: 'Verifier', model: 'verifier-model', instructions: '', tools: [] },
    toolOwnership: ownership, deps: { settings, logger, sessionContextService: context },
  });
  const runtime = createSessionRuntime({ sessionId: 'retry-verifier', agentClient: client, toolOwnership: ownership, deps: { logger, settingsService: settings, sessionContextService: context } });
  try {
    const observer = runtime.turns.start('retry me')[Symbol.asyncIterator]();
    let retryEvent: any;
    for (let count = 0; count < 12; count++) {
      const next = await observer.next();
      if (next.done) break;
      if (next.value.type === 'retry') { retryEvent = next.value; break; }
    }
    expect(retryEvent?.delayMs).toBeGreaterThan(0);
    expect(requests).toBe(1);
    // Let recovery enter the actual backoff instead of only cancelling its emitted event.
    await new Promise((resolve) => setTimeout(resolve, 30));
    await Promise.race([
      runtime.turns.abort(),
      new Promise<never>((_resolve, reject) => setTimeout(() => reject(new Error('backoff abort timed out')), 300)),
    ]);
    await observer.return?.();
    expect(requests).toBe(1);
    const successor = [];
    for await (const next of runtime.turns.start('successor')) successor.push(next);
    expect(JSON.stringify(successor)).toContain('RETRY_SUCCESSOR');
    expect(requests).toBe(2);
  } finally {
    client.abort(); runtime.dispose(); unregisterProvider(providerId); await rm(root, { recursive: true, force: true });
  }
});
